local function addToggle(name, key, y)
    local b = makeButton(
        Content,
        "",
        UDim2.fromOffset(5, y),
        UDim2.new(1, -15, 0, 37)
    )

    local function refresh()
        b.Text = name .. " : " ..
            (Config[key] and "ON" or "OFF")

        b.BackgroundColor3 = Config[key]
            and Color3.fromRGB(0, 105, 105)
            or Color3.fromRGB(35, 43, 58)
    end

    refresh()

    connect(b.MouseButton1Click, function()
        Config[key] = not Config[key]
        refresh()
    end)
end

local function addSlider(name, key, min, max, y)
    local label = Instance.new("TextLabel")
    label.Position = UDim2.fromOffset(5, y)
    label.Size = UDim2.new(1, -15, 0, 24)
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextXAlignment = Enum.TextXAlignment.Left
    label.Font = Enum.Font.Gotham
    label.TextSize = 13
    label.Parent = Content

    local bar = Instance.new("TextButton")
    bar.Position = UDim2.fromOffset(5, y + 30)
    bar.Size = UDim2.new(1, -15, 0, 8)
    bar.Text = ""
    bar.BackgroundColor3 = Color3.fromRGB(45, 52, 68)
    bar.Parent = Content

    Instance.new("UICorner", bar).CornerRadius = UDim.new(1, 0)

    local fill = Instance.new("Frame")
    fill.BackgroundColor3 = Config.ESPColor
    fill.BorderSizePixel = 0
    fill.Parent = bar

    Instance.new("UICorner", fill).CornerRadius = UDim.new(1, 0)

    local dragging = false

    local function update(x)
        local alpha = math.clamp(
            (x - bar.AbsolutePosition.X) /
            bar.AbsoluteSize.X, 0, 1
        )

        Config[key] = math.floor(min + alpha * (max - min))

        fill.Size = UDim2.new(alpha, 0, 1, 0)
        label.Text = name .. " : " .. Config[key]
    end

    update(
        bar.AbsolutePosition.X +
        ((Config[key] - min) / (max - min)) *
        bar.AbsoluteSize.X
    )

    connect(bar.InputBegan, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = true
            update(input.Position.X)
        end
    end)

    connect(UserInputService.InputChanged, function(input)
        if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
            update(input.Position.X)
        end
    end)

    connect(UserInputService.InputEnded, function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging = false
        end
    end)
end

local function getNPCs()
    local result = {}

    for _, model in ipairs(Workspace:GetDescendants()) do
        if model:IsA("Model") then
            local humanoid = model:FindFirstChildOfClass("Humanoid")
            local root = model:FindFirstChild("HumanoidRootPart")

            if humanoid and root
                and humanoid.Health > 0
                and not Players:GetPlayerFromCharacter(model)
                and model ~= Player.Character then

                table.insert(result, model)
            end
        end
    end

    return result
end

local function getAimTarget()
    local bestTarget = nil
    local bestDistance = Config.FOV
    local center = Vector2.new(
        Camera.ViewportSize.X / 2,
        Camera.ViewportSize.Y / 2
    )

    for _, npc in ipairs(getNPCs()) do
        local part = npc:FindFirstChild(Config.AimPart)
            or npc:FindFirstChild("Head")

        if part then
            local screen, visible = Camera:WorldToViewportPoint(part.Position)

            if visible then
                local distance = (
                    Vector2.new(screen.X, screen.Y) - center
                ).Magnitude

                if distance < bestDistance then
                    bestDistance = distance
                    bestTarget = part
                end
            end
        end
    end

    return bestTarget
end

local function setLine(line, a, b)
    local delta = b - a
    line.Position = UDim2.fromOffset(a.X, a.Y)
    line.Size = UDim2.fromOffset(delta.Magnitude, Config.LineThickness)
    line.Rotation = math.deg(math.atan2(delta.Y, delta.X))
    line.Visible = true
end

local function createESP(model)
    if ESPObjects[model] then return end

    local folder = Instance.new("Folder")
    folder.Name = "NPC_ESP"
    folder.Parent = ESPLayer

    local box = Instance.new("Frame")
    box.BackgroundTransparency = 1
    box.BorderSizePixel = 0
    box.Parent = folder

    local boxStroke = Instance.new("UIStroke")
    boxStroke.Color = Config.ESPColor
    boxStroke.Thickness = Config.LineThickness
    boxStroke.Parent = box

    local label = Instance.new("TextLabel")
    label.BackgroundTransparency = 1
    label.TextColor3 = Color3.new(1, 1, 1)
    label.TextStrokeTransparency = 0
    label.Font = Enum.Font.GothamBold
    label.TextSize = 12
    label.Parent = folder

    local healthBack = Instance.new("Frame")
    healthBack.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
    healthBack.BorderSizePixel = 0
    healthBack.Parent = folder

    local healthFill = Instance.new("Frame")
    healthFill.BackgroundColor3 = Color3.fromRGB(40, 220, 100)
    healthFill.BorderSizePixel = 0
    healthFill.Parent = healthBack

    local skeleton = {}
    for i = 1, 14 do
        skeleton[i] = makeLine(folder)
    end

    ESPObjects[model] = {
        Folder = folder,
        Box = box,
        BoxStroke = boxStroke,
        Label = label,
        HealthBack = healthBack,
        HealthFill = healthFill,
        Skeleton = skeleton
    }
end

local skeletonConnections = {
    {"Head", "UpperTorso"},
    {"UpperTorso", "LowerTorso"},
    {"UpperTorso", "LeftUpperArm"},
    {"LeftUpperArm", "LeftLowerArm"},
    {"LeftLowerArm", "LeftHand"},
    {"UpperTorso", "RightUpperArm"},
    {"RightUpperArm", "RightLowerArm"},
    {"RightLowerArm", "RightHand"},
    {"LowerTorso", "LeftUpperLeg"},
    {"LeftUpperLeg", "LeftLowerLeg"},
    {"LeftLowerLeg", "LeftFoot"},
    {"LowerTorso", "RightUpperLeg"},
    {"RightUpperLeg", "RightLowerLeg"},
    {"RightLowerLeg", "RightFoot"}
}

local function updateESP()
    for _, npc in ipairs(getNPCs()) do
        if not ESPObjects[npc] then
            createESP(npc)
        end
    end

    for model, data in pairs(ESPObjects) do
        if not model.Parent then
            data.Folder:Destroy()
            ESPObjects[model] = nil
            continue
        end

        local humanoid = model:FindFirstChildOfClass("Humanoid")
        local root = model:FindFirstChild("HumanoidRootPart")

        if not humanoid or not root then
            data.Folder.Parent = nil
            continue
        end

        local myRoot = Player.Character
            and Player.Character:FindFirstChild("HumanoidRootPart")

        local distance = myRoot
            and (myRoot.Position - root.Position).Magnitude
            or math.huge

        local active = distance <= Config.MaxDistance

        local cf, size = model:GetBoundingBox()
        local corners = {}

        for x = -1, 1, 2 do
            for y = -1, 1, 2 do
                for z = -1, 1, 2 do
                    local world = cf:PointToWorldSpace(
                        Vector3.new(
                            size.X * x / 2,
                            size.Y * y / 2,
                            size.Z * z / 2
                        )
                    )

                    local point, visible = Camera:WorldToViewportPoint(world)
                    if visible then
                        table.insert(corners, Vector2.new(point.X, point.Y))
                    end
                end
            end
        end

        if active and #corners == 8 then
            local minX, minY = math.huge, math.huge
            local maxX, maxY = -math.huge, -math.huge

            for _, p in ipairs(corners) do
                minX = math.min(minX, p.X)
                minY = math.min(minY, p.Y)
                maxX = math.max(maxX, p.X)
                maxY = math.max(maxY, p.Y)
            end

            data.Box.Visible = Config.BoxESP
            data.Box.Position = UDim2.fromOffset(minX, minY)
            data.Box.Size = UDim2.fromOffset(maxX - minX, maxY - minY)
            data.BoxStroke.Color = Config.ESPColor
            data.BoxStroke.Thickness = Config.LineThickness

            data.Label.Visible = Config.NameESP or Config.DistanceESP
                or Config.HealthESP

            local info = {}

            if Config.NameESP then
                table.insert(info, model.Name)
            end

            if Config.DistanceESP then
                table.insert(info, math.floor(distance) .. "m")
            end

            if Config.HealthESP then
                table.insert(info,
                    math.floor(humanoid.Health) .. "/" ..
                    math.floor(humanoid.MaxHealth) .. " HP"
                )
            end

            data.Label.Text = table.concat(info, " | ")
            data.Label.Position = UDim2.fromOffset(minX, minY - 20)
            data.Label.Size = UDim2.fromOffset(maxX - minX, 18)

            data.HealthBack.Visible = Config.HealthESP
            data.HealthBack.Position = UDim2.fromOffset(minX - 6, minY)
            data.HealthBack.Size = UDim2.fromOffset(4, maxY - minY)

            local ratio = humanoid.MaxHealth > 0
                and math.clamp(humanoid.Health / humanoid.MaxHealth, 0, 1)
                or 0

            data.HealthFill.Size = UDim2.new(1, 0, ratio, 0)
            data.HealthFill.Position = UDim2.new(0, 0, 1 - ratio, 0)

            local parts = {}
            for _, pair in ipairs(skeletonConnections) do
                local a = model:FindFirstChild(pair[1])
                local b = model:FindFirstChild(pair[2])

                if a and b then
                    local p1, v1 = Camera:WorldToViewportPoint(a.Position)
                    local p2, v2 = Camera:WorldToViewportPoint(b.Position)

                    if v1 and v2 then
                        table.insert(parts, {
                            Vector2.new(p1.X, p1.Y),
                            Vector2.new(p2.X, p2.Y)
                        })
                    end
                end
            end

            for i, line in ipairs(data.Skeleton) do
                local pair = parts[i]

                line.Visible = Config.SkeletonESP and pair ~= nil
                line.BackgroundColor3 = Config.ESPColor

                if pair then
                    setLine(line, pair[1], pair[2])
                end
            end
        else
            data.Box.Visible = false
            data.Label.Visible = false
            data.HealthBack.Visible = false

            for _, line in ipairs(data.Skeleton) do
                line.Visible = false
            end
        end
    end
end

connect(RunService.RenderStepped, function(dt)
    if not alive then return end

    Camera = Workspace.CurrentCamera

    FOVCircle.Visible = Config.AimAssist
    FOVCircle.Size = UDim2.fromOffset(Config.FOV * 2, Config.FOV * 2)
    FOVStroke.Color = Config.ESPColor

    updateESP()

    if Config.AimAssist and aiming then
        local target = getAimTarget()

        if target then
            local screen = Camera:WorldToViewportPoint(target.Position)

            local center = Vector2.new(
                Camera.ViewportSize.X / 2,
                Camera.ViewportSize.Y / 2
            )

            local targetPoint = Vector2.new(screen.X, screen.Y)
            local delta = targetPoint - center

            local smooth = math.max(1, Config.Smoothness)
            local factor = math.clamp((100 - smooth) / 100, 0.02, 1)

            local newPoint = center + delta * factor

            local ray = Camera:ViewportPointToRay(
                newPoint.X,
                newPoint.Y
            )

            -- Assistência de câmera apenas no ambiente de treino.
            local targetCFrame = CFrame.lookAt(
                Camera.CFrame.Position,
                target.Position
            )

            Camera.CFrame = Camera.CFrame:Lerp(
                targetCFrame,
                math.clamp(dt * (11 - smooth / 12), 0.02, 0.8)
            )
        end
    end
end)
